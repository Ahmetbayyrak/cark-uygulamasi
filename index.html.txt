<script>
const CATS={
 '🍕 Ne Yiyeceğiz?':['Pizza','Burger','Kebap','Sushi','Makarna','Döner','Lahmacun','Tavuk Wrap'],
 '🎬 Ne İzleyeceğiz?':['Komedi','Aksiyon','Korku','Bilim Kurgu','Dizi Maratonu','Belgesel','Animasyon','Romantik'],
 '🎯 Bugün Ne Yapalım?':['Sinema','Yürüyüş','Bowling','Oyun Gecesi','Piknik','Müze','Karaoke','Evde Dinlen'],
 '☕ Kahve/Mola':['Filtre Kahve','Latte','Türk Kahvesi','Espresso','Çay','Mocha','Cold Brew','Limonata']
};
const COLORS=['#8b5cf6','#06b6d4','#ec4899','#f59e0b','#10b981','#3b82f6','#f43f5e','#a3e635','#d946ef','#14b8a6'];
const RULES=['Çarkın kararı kesindir. Temyiz mahkemesi evrenin ta kendisidir.','Burada itiraz kabul edilmez. Yıldızlar böyle yazdı.','Çark konuştu, sen sustun. Bu antlaşma geri alınamaz.','İtiraz dilekçeleri anında kara deliğe gönderilir.','Kader mührünü bastı. Şikayet hattı kapalıdır.'];
const $=id=>document.getElementById(id);
let items=[],rot=0,spinning=false,lastIdx=-1;
try{items=JSON.parse(localStorage.getItem('wheelItems'))||[]}catch(e){}
if(!items.length)items=[...CATS['🍕 Ne Yiyeceğiz?']];
const save=()=>{try{localStorage.setItem('wheelItems',JSON.stringify(items))}catch(e){}};

// ---------- Audio ----------
let ac;
const A=()=>ac||(ac=new (window.AudioContext||window.webkitAudioContext)());
function tone(f,t0,d,type='sine',v=.15){
  const c=A(),o=c.createOscillator(),g=c.createGain();
  o.type=type;o.frequency.value=f;
  g.gain.setValueAtTime(v,c.currentTime+t0);
  g.gain.exponentialRampToValueAtTime(.0001,c.currentTime+t0+d);
  o.connect(g).connect(c.destination);o.start(c.currentTime+t0);o.stop(c.currentTime+t0+d+.02);
}
const tick=()=>tone(700+Math.random()*120,0,.04,'square',.08);
function fanfare(){[523,659,784,1047,1319].forEach((f,i)=>tone(f,i*.11,.35,'triangle',.18));tone(1568,.6,.8,'sine',.15)}

// ---------- Wheel ----------
const cv=$('wheel'),ctx=cv.getContext('2d');
function size(){const d=devicePixelRatio||1,s=cv.clientWidth;cv.width=cv.height=s*d;ctx.setTransform(d,0,0,d,0,0);draw()}
function draw(){
  const s=cv.clientWidth,r=s/2,n=items.length,a=2*Math.PI/n;
  ctx.clearRect(0,0,s,s);
  ctx.save();ctx.translate(r,r);ctx.rotate(rot);
  items.forEach((t,i)=>{
    const c=COLORS[i%COLORS.length],g=ctx.createRadialGradient(0,0,r*.1,0,0,r);
    g.addColorStop(0,c+'cc');g.addColorStop(1,c);
    ctx.beginPath();ctx.moveTo(0,0);ctx.arc(0,0,r-6,i*a,(i+1)*a);ctx.closePath();
    ctx.fillStyle=g;ctx.fill();ctx.strokeStyle='rgba(255,255,255,.35)';ctx.lineWidth=2;ctx.stroke();
    ctx.save();ctx.rotate(i*a+a/2);ctx.textAlign='right';ctx.fillStyle='#fff';
    ctx.font=`700 ${Math.max(11,Math.min(22,s/(n>12?30:20)))}px Outfit`;
    ctx.shadowColor='rgba(0,0,0,.6)';ctx.shadowBlur=4;
    let txt=t.length>16?t.slice(0,15)+'…':t;
    ctx.fillText(txt,r-24,6);ctx.restore();
  });
  ctx.restore();
  // hub
  ctx.beginPath();ctx.arc(r,r,s*.07,0,7);
  const h=ctx.createRadialGradient(r,r,2,r,r,s*.07);h.addColorStop(0,'#fff');h.addColorStop(1,'#a78bfa');
  ctx.fillStyle=h;ctx.shadowColor='#8b5cf6';ctx.shadowBlur=20;ctx.fill();
}
const idxAt=()=>{const a=2*Math.PI/items.length;let x=(-Math.PI/2-rot)%(2*Math.PI);if(x<0)x+=2*Math.PI;return Math.floor(x/a)%items.length};

function spin(){
  if(spinning||items.length<2)return;
  A().resume();spinning=true;$('spin').disabled=true;
  const start=rot,total=Math.PI*2*(6+Math.random()*4)+Math.random()*Math.PI*2,dur=5500+Math.random()*1500,t0=performance.now();
  lastIdx=idxAt();
  const ease=p=>p<.08?(p/.08)*(p/.08)*.02*0+ (p/.08)*(p/.08)*.08:.08+.92*(1-Math.pow(1-(p-.08)/.92,4));
  (function f(now){
    const p=Math.min(1,(now-t0)/dur);
    rot=start+total*ease(p);
    const i=idxAt();if(i!==lastIdx){lastIdx=i;tick()}
    draw();
    if(p<1)requestAnimationFrame(f);else end();
  })(t0);
}
function end(){
  spinning=false;$('spin').disabled=false;
  $('win').textContent=items[idxAt()];
  $('rule').textContent=RULES[Math.floor(Math.random()*RULES.length)];
  $('modal').classList.add('show');
  fanfare();burst();
}

// ---------- Confetti ----------
const fx=$('fx'),fc=fx.getContext('2d');let parts=[],anim=false;
function fsize(){fx.width=innerWidth;fx.height=innerHeight}
function burst(){
  for(let k=0;k<2;k++)for(let i=0;i<120;i++){
    const ang=Math.random()*Math.PI*2,sp=6+Math.random()*12;
    parts.push({x:innerWidth/2,y:innerHeight/2.4,vx:Math.cos(ang)*sp,vy:Math.sin(ang)*sp-6,w:6+Math.random()*8,h:4+Math.random()*6,
      c:COLORS[Math.floor(Math.random()*COLORS.length)],r:Math.random()*6,vr:(Math.random()-.5)*.4,life:140+Math.random()*80});
  }
  if(!anim){anim=true;loop()}
}
function loop(){
  fc.clearRect(0,0,fx.width,fx.height);
  parts=parts.filter(p=>p.life>0&&p.y<fx.height+30);
  parts.forEach(p=>{p.vy+=.25;p.vx*=.99;p.x+=p.vx;p.y+=p.vy;p.r+=p.vr;p.life--;
    fc.save();fc.translate(p.x,p.y);fc.rotate(p.r);fc.fillStyle=p.c;fc.globalAlpha=Math.min(1,p.life/40);fc.fillRect(-p.w/2,-p.h/2,p.w,p.h);fc.restore()});
  if(parts.length)requestAnimationFrame(loop);else{anim=false;fc.clearRect(0,0,fx.width,fx.height)}
}

function render(){
  $('cnt').textContent=items.length;
  $('list').innerHTML='';
  items.forEach((t,i)=>{
    const d=document.createElement('div');d.className='item';
    d.innerHTML=`<span><i style="display:inline-block;width:10px;height:10px;border-radius:50%;background:${COLORS[i%COLORS.length]};margin-right:8px"></i></span>`;
    d.querySelector('span').append(document.createTextNode(t));
    const x=document.createElement('span');x.className='x';x.textContent='✕';
    x.onclick=()=>{if(spinning)return;items.splice(i,1);update()};
    d.append(x);$('list').append(d);
  });
  $('spin').disabled=spinning||items.length<2;
  draw();
}
function update(){save();render()}
function addItem(){
  const v=$('inp').value.trim();if(!v||spinning)return;
  items.push(v);$('inp').value='';update();$('inp').focus();
}
Object.keys(CATS).forEach(k=>{
  const b=document.createElement('span');b.className='chip';b.textContent=k;
  b.onclick=()=>{if(spinning)return;items=[...CATS[k]];document.querySelectorAll('#cats .chip').forEach(c=>c.classList.remove('on'));b.classList.add('on');update()};
  $('cats').append(b);
});

$('inp').onkeydown=e=>{if(e.key==='Enter')addItem()};
